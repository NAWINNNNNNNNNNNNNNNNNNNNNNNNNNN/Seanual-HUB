# -----------------------------
# Helper: Ensure SysWOW64 exists
# -----------------------------
function Ensure-SysWOW64 {
    if (-not (Test-Path $sysWOW64)) {
        Write-Host "[!] SysWOW64 folder not found. Creating..." -ForegroundColor Yellow
        try {
            New-Item -ItemType Directory -Path $sysWOW64 -Force | Out-Null
            Write-Host "[✓] Created SysWOW64 folder." -ForegroundColor Green
        } catch {
            Write-Host "[✗] Failed to create SysWOW64 folder: $_" -ForegroundColor Red
            Write-Host "`nPress any key to exit..."
            Read-Host
            exit
        }
    }
}

# -----------------------------
# Helper: Download file to destination
# -----------------------------
function Download-ToSysWOW64 {
    param(
        [string]$Url,
        [string]$FileName
    )

    $dest = Join-Path $sysWOW64 $FileName
    $temp = Join-Path $env:TEMP ("dl_" + [System.Guid]::NewGuid().ToString() + "_" + $FileName)

    Write-Host "[+] Downloading $FileName ..." -ForegroundColor Cyan
    Write-Host "[+] Target: $dest" -ForegroundColor Gray

    try {
        Invoke-WebRequest -Uri $Url -OutFile $temp -ErrorAction Stop

        if (Test-Path $dest) {
            Remove-Item $dest -Force -ErrorAction SilentlyContinue
        }

        Move-Item -Path $temp -Destination $dest -Force -ErrorAction Stop
        Write-Host "[✓] $FileName downloaded successfully!" -ForegroundColor Green
        return $true
    } catch {
        Write-Host "[✗] Failed to download $FileName : $_" -ForegroundColor Red
        if (Test-Path $temp) {
            Remove-Item $temp -Force -ErrorAction SilentlyContinue
        }
        return $false
    }
}
